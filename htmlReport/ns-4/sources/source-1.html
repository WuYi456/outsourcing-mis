 package com.tx.outsourcingmis.config;
 
 import co.elastic.clients.elasticsearch.ElasticsearchClient;
 import lombok.extern.slf4j.Slf4j;
 import org.springframework.boot.actuate.health.Health;
 import org.springframework.boot.actuate.health.HealthIndicator;
 import org.springframework.stereotype.Component;
 
 import java.io.IOException;
 
 /**
  * 自定义 Elasticsearch 健康检查
  * 使用手动配置的 ElasticsearchClient（已跳过SSL验证）
  *
  */
 @Slf4j
 @Component
 public class CustomElasticsearchHealthIndicator implements HealthIndicator {
 
     private final ElasticsearchClient elasticsearchClient;
 
     public CustomElasticsearchHealthIndicator(ElasticsearchClient elasticsearchClient) {
         this.elasticsearchClient = elasticsearchClient;
     }
 
     @Override
     public Health health() {
         // 检查客户端是否初始化成功
         if (elasticsearchClient == null) {
             log.warn("Elasticsearch客户端未初始化，跳过健康检查");
             return Health.unknown()
                     .withDetail("reason", "ElasticsearchClient未初始化")
                     .withDetail("status", "client_not_available")
                     .build();
         }
 
         try {
             // 尝试执行 ping 操作
             boolean isConnected = elasticsearchClient.ping().value();
 
             if (isConnected) {
                 log.debug("Elasticsearch健康检查通过");
 
                 // 获取ES版本信息（可选）
                 String version = null;
                 try {
                     var info = elasticsearchClient.info();
                     version = info.version().number();
                 } catch (Exception e) {
                     log.debug("获取ES版本信息失败: {}", e.getMessage());
                 }
 
                 Health.Builder builder = Health.up()
                         .withDetail("service", "Elasticsearch")
                         .withDetail("status", "connected")
                         .withDetail("using", "custom_client_with_ssl_disabled");
 
                 if (version != null) {
                     builder.withDetail("version", version);
                 }
 
                 return builder.build();
             } else {
                 log.warn("Elasticsearch ping 返回 false");
                 return Health.down()
                         .withDetail("reason", "Ping 返回 false")
                         .withDetail("status", "not_responding")
                         .build();
             }
 
         } catch (IOException e) {
             log.error("Elasticsearch健康检查失败: {}", e.getMessage());
             return Health.down()
                     .withDetail("reason", e.getMessage())
                     .withDetail("error_type", e.getClass().getSimpleName())
                     .withDetail("status", "connection_error")
                     .build();
         } catch (Exception e) {
             log.error("Elasticsearch健康检查异常: {}", e.getMessage());
             return Health.down()
                     .withDetail("reason", "Unexpected error: " + e.getMessage())
                     .withDetail("error_type", e.getClass().getSimpleName())
                     .build();
         }
     }
 }
