 package com.tx.outsourcingmis.service.impl;
 
 import com.tx.outsourcingmis.dto.OperationLogQueryRequest;
 import com.tx.outsourcingmis.dto.OperationLogResponse;
 import com.tx.outsourcingmis.es.document.OperationLogDocument;
 import com.tx.outsourcingmis.es.repository.OperationLogElasticsearchRepository;
 import com.tx.outsourcingmis.service.OperationLogService;
 import lombok.RequiredArgsConstructor;
 import lombok.extern.slf4j.Slf4j;
 import org.springframework.data.domain.Page;
 import org.springframework.data.domain.PageRequest;
 import org.springframework.data.domain.Pageable;
 import org.springframework.data.domain.Sort;
 import org.springframework.scheduling.annotation.Async;
 import org.springframework.stereotype.Service;
 
 import java.time.LocalDateTime;
 import java.time.format.DateTimeFormatter;
 
 @Slf4j
 @Service
 @RequiredArgsConstructor
 public class OperationLogServiceImpl implements OperationLogService {
 
     private final OperationLogElasticsearchRepository esRepository;
 
     private static final DateTimeFormatter DATE_TIME_FORMATTER =
             DateTimeFormatter.ofPattern("yyyy-MM-dd HH:mm:ss");
 
     @Override
     @Async("taskExecutor")
     public void saveLogAsync(OperationLogDocument logDoc) {
         saveLog(logDoc);
     }
 
     @Override
     public void saveLog(OperationLogDocument logDoc) {
         // 检查ES仓库是否可用
         if (esRepository == null) {
             log.warn("⚠️ Elasticsearch不可用，操作日志跳过保存: {} - {}",
                     logDoc != null ? logDoc.getOperation() : "null",
                     logDoc != null ? logDoc.getUsername() : "unknown");
             return;
         }
 
         try {
             if (logDoc == null) {
                 log.warn("操作日志对象为null，跳过保存");
                 return;
             }
 
             if (logDoc.getCreateTime() == null) {
                 logDoc.setCreateTime(LocalDateTime.now());
             }
 
             esRepository.save(logDoc);
             log.debug("✅ 操作日志已存入ES: {} - {}", logDoc.getUsername(), logDoc.getOperation());
         } catch (Exception e) {
             log.error("❌ 操作日志存入ES失败: {}", e.getMessage(), e);
         }
     }
 
     @Override
     public OperationLogDocument getById(String id) {
         if (esRepository == null) {
             log.warn("Elasticsearch不可用，无法查询日志");
             return null;
         }
         try {
             return esRepository.findById(id).orElse(null);
         } catch (Exception e) {
             log.error("查询日志失败: {}", e.getMessage(), e);
             return null;
         }
     }
 
     @Override
     public Page<OperationLogResponse> getByUserId(Long userId, Integer pageNum, Integer pageSize) {
         if (esRepository == null) {
             log.warn("Elasticsearch不可用，返回空结果");
             return Page.empty();
         }
         // ES中按userId查询需要特殊处理，这里用username替代
         // 实际场景中userId和username可以都存，这里简化按username查
         return Page.empty();
     }
 
     @Override
     public Page<OperationLogResponse> getByUsername(String username, Integer pageNum, Integer pageSize) {
         if (esRepository == null) {
             log.warn("Elasticsearch不可用，返回空结果");
             return Page.empty();
         }
         try {
             Pageable pageable = PageRequest.of(pageNum - 1, pageSize, Sort.by("createTime").descending());
             Page<OperationLogDocument> page = esRepository.findByUsername(username, pageable);
             return page.map(this::convertToResponse);
         } catch (Exception e) {
             log.error("按用户名查询日志失败: {}", e.getMessage(), e);
             return Page.empty();
         }
     }
 
     @Override
     public Page<OperationLogResponse> getByOperation(String operation, Integer pageNum, Integer pageSize) {
         if (esRepository == null) {
             log.warn("Elasticsearch不可用，返回空结果");
             return Page.empty();
         }
         try {
             Pageable pageable = PageRequest.of(pageNum - 1, pageSize, Sort.by("createTime").descending());
             Page<OperationLogDocument> page = esRepository.findByOperation(operation, pageable);
             return page.map(this::convertToResponse);
         } catch (Exception e) {
             log.error("按操作类型查询日志失败: {}", e.getMessage(), e);
             return Page.empty();
         }
     }
 
     @Override
     public Page<OperationLogResponse> search(OperationLogQueryRequest request) {
         if (esRepository == null) {
             log.warn("Elasticsearch不可用，返回空结果");
             return Page.empty();
         }
 
         try {
             Pageable pageable = PageRequest.of(
                     request.getPageNum() - 1,
                     request.getPageSize(),
                     Sort.by("createTime").descending()
             );
 
             Page<OperationLogDocument> page;
 
             // 转换日期格式（支持空格格式和T格式）
             String startTime = convertToISOFormat(request.getStartTime());
             String endTime = convertToISOFormat(request.getEndTime());
 
             boolean hasUsername = request.getUsername() != null && !request.getUsername().isEmpty();
             boolean hasOperation = request.getOperation() != null && !request.getOperation().isEmpty();
             boolean hasTimeRange = startTime != null && endTime != null;
 
             // 根据条件组合查询
             if (hasUsername && hasOperation && hasTimeRange) {
                 // 三个条件都有
                 page = esRepository.searchLogs(
                         request.getUsername(),
                         request.getOperation(),
                         startTime,
                         endTime,
                         pageable
                 );
             } else if (hasUsername && hasOperation) {
                 // 只有用户名和操作类型
                 page = esRepository.findByUsernameAndOperation(
                         request.getUsername(),
                         request.getOperation(),
                         pageable
                 );
             } else if (hasUsername && hasTimeRange) {
                 // 用户名 + 时间范围
                 page = esRepository.searchLogs(
                         request.getUsername(),
                         null,
                         startTime,
                         endTime,
                         pageable
                 );
             } else if (hasOperation && hasTimeRange) {
                 // 操作类型 + 时间范围
                 page = esRepository.searchLogs(
                         null,
                         request.getOperation(),
                         startTime,
                         endTime,
                         pageable
                 );
             } else if (hasUsername) {
                 // 只有用户名
                 page = esRepository.findByUsername(request.getUsername(), pageable);
             } else if (hasOperation) {
                 // 只有操作类型
                 page = esRepository.findByOperation(request.getOperation(), pageable);
             } else if (hasTimeRange) {
                 // 只有时间范围
                 LocalDateTime startDateTime = LocalDateTime.parse(startTime);
                 LocalDateTime endDateTime = LocalDateTime.parse(endTime);
                 page = esRepository.findByCreateTimeBetween(startDateTime, endDateTime, pageable);
             } else {
                 // 无条件，查询全部
                 page = esRepository.findAll(pageable);
             }
 
             return page.map(this::convertToResponse);
         } catch (Exception e) {
             log.error("高级搜索日志失败: {}", e.getMessage(), e);
             return Page.empty();
         }
     }
 
     @Override
     public void deleteById(String id) {
         if (esRepository == null) {
             log.warn("Elasticsearch不可用，无法删除日志");
             return;
         }
         try {
             esRepository.deleteById(id);
             log.info("✅ 删除操作日志成功: id={}", id);
         } catch (Exception e) {
             log.error("❌ 删除操作日志失败: {}", e.getMessage(), e);
         }
     }
 
     /**
      * 将日期字符串转换为 ISO 格式
      * 输入: "2026-06-08 00:00:00" 或 "2026-06-08T00:00:00"
      * 输出: "2026-06-08T00:00:00"
      */
     private String convertToISOFormat(String dateStr) {
         if (dateStr == null || dateStr.isEmpty()) {
             return null;
         }
         if (dateStr.contains("T")) {
             return dateStr;
         }
         return dateStr.replace(" ", "T");
     }
 
     private OperationLogResponse convertToResponse(OperationLogDocument doc) {
         if (doc == null) {
             return null;
         }
         return OperationLogResponse.builder()
                 .id(doc.getId())
                 .userId(doc.getUserId())
                 .username(doc.getUsername())
                 .operation(doc.getOperation())
                 .method(doc.getMethod())
                 .params(doc.getParams())
                 .result(doc.getResult())
                 .ip(doc.getIp())
                 .duration(doc.getDuration())
                 .success(doc.getSuccess())
                 .errorMsg(doc.getErrorMsg())
                 .createTime(doc.getCreateTime())
                 .build();
     }
