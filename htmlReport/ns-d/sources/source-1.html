 package com.tx.outsourcingmis.service.impl;
 
 import com.tx.outsourcingmis.dto.LoginResponse;
 import com.tx.outsourcingmis.entity.User;
 import com.tx.outsourcingmis.mapper.UserMapper;
 import com.tx.outsourcingmis.service.CacheService;
 import lombok.RequiredArgsConstructor;
 import lombok.extern.slf4j.Slf4j;
 import org.springframework.beans.factory.annotation.Value;
 import org.springframework.data.redis.core.RedisTemplate;
 import org.springframework.stereotype.Service;
 
 import java.util.concurrent.TimeUnit;
 
 @Slf4j
 @Service
 @RequiredArgsConstructor
 public class CacheServiceImpl implements CacheService {
 
     private final RedisTemplate<String, Object> redisTemplate;
     private final UserMapper userMapper;
 
     @Value("${redis.cache.user-prefix:user:}")
     private String userPrefix;
 
     @Value("${redis.cache.token-prefix:token:}")
     private String tokenPrefix;
 
     @Value("${redis.cache.session-ttl:7200}")
     private Long sessionTtl;
 
     private String getUserKey(Long userId) {
         return userPrefix + userId;
     }
 
     private String getTokenKey(String token) {
         return tokenPrefix + token;
     }
 
     private String getUserTokenKey(Long userId) {
         return userPrefix + userId + ":token";
     }
 
     @Override
     public void cacheUserLogin(String token, LoginResponse loginResponse, Long userId) {
         String tokenKey = getTokenKey(token);
         String userTokenKey = getUserTokenKey(userId);
 
         // 缓存Token对应的登录信息
         redisTemplate.opsForValue().set(tokenKey, loginResponse, sessionTtl, TimeUnit.SECONDS);
 
         // 缓存用户ID与Token的映射（用于登出时删除）
         redisTemplate.opsForValue().set(userTokenKey, token, sessionTtl, TimeUnit.SECONDS);
 
         log.debug("缓存用户登录信息 - userId: {}, token前10位: {}", userId, token.substring(0, Math.min(10, token.length())));
     }
 
     @Override
     public LoginResponse getLoginResponseByToken(String token) {
         String tokenKey = getTokenKey(token);
         Object cached = redisTemplate.opsForValue().get(tokenKey);
 
         if (cached instanceof LoginResponse) {
             // 刷新过期时间
             redisTemplate.expire(tokenKey, sessionTtl, TimeUnit.SECONDS);
             return (LoginResponse) cached;
         }
         return null;
     }
 
     @Override
     public User getUserById(Long userId) {
         String userKey = getUserKey(userId);
 
         // 先从缓存获取
         Object cached = redisTemplate.opsForValue().get(userKey);
         if (cached instanceof User) {
             log.debug("从Redis获取用户信息 - userId: {}", userId);
             return (User) cached;
         }
 
         // 缓存未命中，从数据库查询
         User user = userMapper.selectById(userId);
         if (user != null) {
             cacheUser(user);
         }
         return user;
     }
 
     @Override
     public void cacheUser(User user) {
         String userKey = getUserKey(user.getId());
         redisTemplate.opsForValue().set(userKey, user, sessionTtl, TimeUnit.SECONDS);
         log.debug("缓存用户信息 - userId: {}", user.getId());
     }
 
     @Override
     public void removeUserLogin(String token) {
         if (token == null) return;
 
         String tokenKey = getTokenKey(token);
         LoginResponse loginResponse = getLoginResponseByToken(token);
 
         if (loginResponse != null) {
             String userTokenKey = getUserTokenKey(loginResponse.getUserId());
             redisTemplate.delete(userTokenKey);
         }
 
         redisTemplate.delete(tokenKey);
         log.debug("清除用户登录缓存 - token前10位: {}", token.substring(0, Math.min(10, token.length())));
     }
 
     @Override
     public void removeUserCache(Long userId) {
         String userKey = getUserKey(userId);
         redisTemplate.delete(userKey);
         log.debug("清除用户信息缓存 - userId: {}", userId);
     }
 
     @Override
     public boolean isTokenValid(String token) {
         String tokenKey = getTokenKey(token);
         Boolean hasKey = redisTemplate.hasKey(tokenKey);
         return Boolean.TRUE.equals(hasKey);
     }
 
     @Override
     public String getTokenByUserId(Long userId) {
         String userTokenKey = getUserTokenKey(userId);
         Object token = redisTemplate.opsForValue().get(userTokenKey);
         return token instanceof String ? (String) token : null;
     }
 
     @Override
     public void updateUserToken(Long userId, String oldToken, String newToken) {
         // 删除旧Token缓存
         if (oldToken != null) {
             redisTemplate.delete(getTokenKey(oldToken));
         }
 
         // 更新新Token缓存
         String userTokenKey = getUserTokenKey(userId);
         redisTemplate.opsForValue().set(userTokenKey, newToken, sessionTtl, TimeUnit.SECONDS);
 
         log.debug("更新用户Token - userId: {}", userId);
     }
 }
